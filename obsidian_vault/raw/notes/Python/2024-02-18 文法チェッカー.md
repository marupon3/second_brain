# 文法チェッカー

## blackを使って文法をチェックする

## 機能

* 修正が必要する場合、自動的にgitに登録されコメントを付ける
* 文法に問題ない場合、エラーがあっても修正しない場合はそのまま

<!-- 変換時の注記: OneNote由来の改行を括弧深さ/構文解析で自動復元しました。インデントはOneNote側で失われている場合があり保証されません。実行前に必ず目視確認してください（要確認）。 -->

```python
import tkinter as tk
from tkinter import filedialog
from tkinter import messagebox
import subprocess
import sys
def select_file_and_check_format():
    root = tk.Tk()
    root.withdraw()  # Tkinterのルートウィンドウを表示しない
    file_path = filedialog.askopenfilename( title="Pythonファイルを選択してください", filetypes=[("Pythonファイル", "*.py")] )
    if file_path:
        print(f"選択されたファイル：{file_path}")
        # blackを使用してファイルのフォーマットをチェック
        result = subprocess.run( [sys.executable, "-m", "black", "--check", file_path], capture_output=True, text=True, encoding='utf-8', errors='ignore' )
        if result.returncode != 0:
            print("フォーマットエラーが検出されました。")
            # ユーザーに修正するかどうかを尋ねる
            if messagebox.askyesno("フォーマットエラー", "ファイルのフォーマットを修正しますか？"):
                # 修正を選択した場合、Gitに追加し、コミットする
                subprocess.run(["git", "add", file_path])
                subprocess.run(["git", "commit", "-m", "black修正前コミット"])
                # ファイルを修正する
                subprocess.run([sys.executable, "-m", "black", file_path])
                messagebox.showinfo("修正完了", "ファイルが修正されました。")
                print("ファイルが修正されました。")
            else:
                print("ファイルの修正はキャンセルされました。")
        else:
            messagebox.showinfo("フォーマットチェック", "ファイルのフォーマットは正しいです。")
            print("ファイルのフォーマットは正しいです。")
    else:
        print("ファイルが選択されませんでした。")
if __name__ == "__main__":
    select_file_and_check_format()
```
