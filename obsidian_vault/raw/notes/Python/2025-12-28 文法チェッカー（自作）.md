import sys
import shutil
def run_check(tool_name, command):
    """コマンドを実行して結果を表示する補助関数"""
    print(f"\n{'='*20} {tool_name} のチェック開始 {'='*20}")
    # ツールがインストールされているか確認
    if not shutil.which(command[0]):
        print(f"警告: {tool_name} がインストールされていないためスキップします。")
        return
    try:
        # コマンド実行
        result = subprocess.run( command, capture_output=True, text=True, encoding='utf-8' )
        # 標準出力と標準エラーを統合して表示
        if result.stdout:
            print(result.stdout)
        if result.stderr:
            print(result.stderr)
        if result.returncode == 0:
            print(f"✅ {tool_name}: 問題は見つかりませんでした！")
        else:
            print(f"⚠️ {tool_name}: いくつかの指摘事項があります。")
    except Exception as e:
        print(f"実行中にエラーが発生しました: {e}")
def main():
    if len(sys.argv) < 2:
        print("使用法: python code_checker.py <チェックしたいファイル名.py>")
        return
    target_file = sys.argv[1]
    # 各ツールの実行設定
    checks = [ ("Ruff (高速チェック/整形)", ["ruff", "check", target_file]), ("Flake8 (PEP8準拠チェック)", ["flake8", target_file]), ("Pylint (コード品質スコアリング)", ["pylint", target_file]), ("mypy (静的型チェック)", ["mypy", target_file]), ]
    print(f"🔍 ターゲットファイル: {target_file}")
    for tool_name, command in checks:
        run_check(tool_name, command)
    print(f"\n{'='*20} すべての診断が終了しました {'='*20}")
if __name__ == "__main__":
    main()
