# _項目をカンマ区切りで出力

SELECT
string_agg(original_word, ',' ORDER BY word_id ASC) AS csv_text FROM ( SELECT original_word, word_id
FROM public.word_dictionary ORDER BY word_id ASC LIMIT 200 ) AS t;
テーブル名: word_dictionary
昇順を設定するカラム: word_id
出力したいカラム: original_word
