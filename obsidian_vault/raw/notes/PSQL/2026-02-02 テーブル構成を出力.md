# _テーブル構成を出力

SELECT
table_name,
column_name,
data_type,
is_nullable,
column_default
FROM
information_schema.columns
WHERE
table_schema = 'public'
AND
table_name IN (
'access_logs',
'crawl_schedules',
'document_metadata',
'inverted_index',
'search_targets',
'textdata',
'user_dictionary',
'user_search_folders',
'word_dictionary'
)
ORDER BY
table_name,
ordinal_position;
